import { chromium } from '~/Repos/atelier/node_modules/playwright/index.mjs';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {createReadStream} from 'node:fs';
import http from 'node:http';import path from 'node:path';
const root='/private/tmp/atelier-complex-ui-20261108';
const out=path.join(root,process.argv[2]??'independent-first');await mkdir(out,{recursive:true});
const server=http.createServer((req,res)=>{const file=path.resolve(root,'.'+new URL(req.url,'http://localhost').pathname);if(!file.startsWith(root+'/'))return res.writeHead(403).end();res.setHeader('Content-Type',file.endsWith('.js')?'text/javascript':file.endsWith('.wasm')?'application/wasm':'text/html');const s=createReadStream(file);s.on('error',()=>res.end());s.pipe(res)});await new Promise(r=>server.listen(0,'127.0.0.1',r));
const browser=await chromium.launch({headless:true,args:['--enable-unsafe-swiftshader']});const results=[];
try {for(const name of ['review','architecture','lab']){const page=await browser.newPage({viewport:{width:1440,height:1000}});const errors=[];page.on('pageerror',e=>errors.push(e.message));page.on('console',e=>{if(e.type()==='error')errors.push(e.text())});await page.goto(`http://127.0.0.1:${server.address().port}/${name}/author/output.html`);await page.waitForFunction(()=>document.documentElement.dataset.artifactReady==='true').catch(()=>{});await page.waitForTimeout(300);let info=await page.evaluate(()=>({ready:document.documentElement.dataset.artifactReady,widgets:[...document.querySelectorAll('[data-at-mounted]')].map(n=>({name:n.dataset.atMounted,state:n.dataset.atState})),cells:[...document.querySelectorAll('.at-cell')].map(n=>({id:n.id,title:n.querySelector('h2')?.textContent})),text:document.body.innerText,overflow:document.documentElement.scrollWidth>innerWidth}));await page.evaluate(()=>scrollTo(0,0));await page.screenshot({path:path.join(out,`${name}-overview.png`)});for(const id of name==='review'?['review','detail','guidance']:name==='architecture'?['workflow','inspector','sequence']:['linked','plot-gallery','training']){const cell=page.locator('#'+id);if(await cell.count()){await cell.screenshot({path:path.join(out,`${name}-${id}.png`)})}}results.push({name,...info,errors});await page.close()}}finally{await browser.close();await new Promise(r=>server.close(r));await writeFile(path.join(out,'probe.json'),JSON.stringify(results,null,2))}console.log(JSON.stringify(results.map(({name,ready,widgets,cells,errors,overflow})=>({name,ready,widgets,cells,errors,overflow})),null,2));
