const icons = Atelier.modules.icons
const book = Atelier.notebook({ title: "Architecture investigation notebook", description: "Illustrative AWS workflow · supplied fixture · no production evidence" })
const intro = book.cell({ id: "brief", title: "Investigation brief" })
const briefStack = Atelier.stack(intro.view)
Atelier.mount("markdown", briefStack, { source: "## A request, an archive, and a review trail\n\nTrace how a request moves through the supplied AWS components, inspect service details, then record a local incident note. The topology is an **illustrative fixture**; it does not represent a live deployment.\n\nUse the left navigation to move through the investigation. Select any node in the graph or service inventory to synchronize the detail inspector." })
Atelier.metrics(briefStack, [
  { label: "Components", value: "12", note: "Supplied fixture" },
  { label: "Connections", value: "12", note: "Illustrative paths" },
  { label: "Health flags", value: "2", note: "Warning badges" },
])
const graphCell = book.cell({ id: "workflow", title: "AWS workflow map", controls: true })
let selectedId = "n1"
let health = "All"
let flow
let detailViews
let summaryHandle, propertiesHandle, exampleHandle
let noteValue = ""
const current = () => fixture.nodes.find(n => n.id === selectedId)
const toneFor = n => health === "All" || n.status === health ? (n.status === "Warning" ? "warning" : "success") : "neutral"
const graphData = () => ({
  nodes: fixture.nodes.map(n => ({ id: n.id, selected: n.id === selectedId, data: { label: n.label, description: n.description, status: n.status, icon: n.icon ? icons.url(n.icon) : undefined, tone: toneFor(n) } })),
  edges: fixture.edges.map(([a,b]) => ({ id: `e${a}-${b}`, source: `n${a}`, target: `n${b}` })),
})
const choose = async (id, focus = false) => {
  selectedId = id
  selectNode.value = id
  if (flow) {
    flow.api.flow?.setNodes(nodes => nodes.map(node => {
      const record = fixture.nodes.find(item => item.id === node.id)
      return { ...node, selected: node.id === selectedId, data: { ...node.data, tone: record ? toneFor(record) : node.data.tone } }
    }))
    if (focus) flow.api.focus(id)
  }
  await refreshDetails()
}
flow = Atelier.mount("flow", graphCell.view, { ...graphData(), onNodeClick: (_event, node) => void choose(node.id) })
await flow.ready
const selectNode = Atelier.field(graphCell.controls, { label: "Selected service", type: "select", value: selectedId, options: fixture.nodes.map(n => ({ value: n.id, label: `${n.id} · ${n.service}` })), onChange: id => void choose(id, true) })
Atelier.field(graphCell.controls, { label: "Health emphasis", type: "select", value: health, options: ["All", "Ready", "Warning"], description: "Highlights matching nodes; topology remains visible.", onChange: value => { health = value; flow.api.flow?.setNodes(nodes => nodes.map(node => { const record = fixture.nodes.find(item => item.id === node.id); return { ...node, data: { ...node.data, tone: record ? toneFor(record) : node.data.tone } } })) } })
Atelier.button("Fit overview", () => flow.api.flow?.fitView({ padding: 0.16 }), graphCell.controls, { variant: "secondary" })
Atelier.button("Focus selected", () => flow.api.focus(selectedId), graphCell.controls, { variant: "secondary" })
Atelier.canvasLink(flow, graphData(), { parent: graphCell.header })

const inspector = book.cell({ id: "inspector", title: "Selected service inspector", controls: true })
const incidentNote = Atelier.field(inspector.controls, { label: "Local incident note", type: "textarea", value: noteValue, description: "Stored in this notebook session and included in the export.", onChange: value => { noteValue = value } })
detailViews = Atelier.tabs(inspector.view, {
  label: "Service detail views",
  items: [
    { id: "summary", label: "Summary", render: panel => { summaryHandle = Atelier.mount("markdown", panel, { source: "" }); return summaryHandle } },
    { id: "properties", label: "Properties", render: panel => { propertiesHandle = Atelier.mount("json", panel, { data: propertiesFor(current()) }); return propertiesHandle } },
    { id: "example", label: "Example", render: panel => { const node = current(); exampleHandle = Atelier.mount("code", panel, { language: node.id === "n12" ? "sql" : "json", source: exampleFor(node) }); return exampleHandle } },
  ],
})
const propertiesFor = node => ({ id: node.id, service: node.service, status: node.status, description: node.description, icon: node.icon ?? "No matching AWS icon in the bundled catalog" })
const exampleFor = node => node.id === "n12" ? `SELECT record_id, decision, archived_at\nFROM evidence_audit\nWHERE service_id = '${node.id}'\nORDER BY archived_at DESC;` : JSON.stringify({ serviceId: node.id, traceId: "trace-example-7f2a", event: "processing.requested", evidenceVersion: "v1" }, null, 2)
async function refreshDetails() {
  const n = current()
  if (summaryHandle) await summaryHandle.update({ source: `## ${n.service}\n\n**${n.status}** · ID ${n.id}\n\n${n.description}\n\nThis service description comes from the supplied fixture. The local note remains available while switching tabs.` })
  if (propertiesHandle) await propertiesHandle.update({ data: propertiesFor(n) })
  if (exampleHandle) await exampleHandle.update({ language: n.id === "n12" ? "sql" : "json", source: exampleFor(n) })
}
Atelier.button("Export selection + note", () => Atelier.download("architecture-investigation.json", JSON.stringify({ selectedId, incidentNote: noteValue }, null, 2)), inspector.header, { variant: "primary" })
await detailViews.ready
await refreshDetails()
