const filtered = () => records.filter((r) => (group === "All" || r.group === group) && r.title.toLowerCase().includes(query.toLowerCase()))
const selected = () => records.find((r) => r.id === selectedId) || records[0]
const chartOptions = (rows) => ({ spec: { data: { values: rows }, width: "container", height: 260, mark: { type: "point", filled: true, size: 110 }, encoding: { x: { field: "score", type: "quantitative", title: "Illustrative score" }, y: { field: "volume", type: "quantitative", title: "Volume" }, color: { field: "group", type: "nominal", title: "Group" }, tooltip: [{ field: "title" }, { field: "group" }, { field: "decision" }, { field: "score" }] } } })
const summaryOptions = () => { const r = selected(); return { source: `## ${r.title}\n\n**Stable record ID:** ${r.id}  \n**Group:** ${r.group} · **Status:** ${r.status} · **Decision:** ${r.decision}\n\n${r.summary}\n\n**Local draft note**\n\n${draftNote || "No note drafted yet."}` } }
async function refreshDetails() {
  const r = selected()
  if (summaryView) await summaryView.update(summaryOptions())
  if (jsonView) await jsonView.update({ data: r })
  if (comparisonView) await comparisonView.update({ type: "compare", before: images[r.imageIndex], after: images[(r.imageIndex + 1) % images.length], beforeAlt: "Selected supplied illustration", afterAlt: "Next supplied illustration", position: 50, controls: detail.controls })
}
function updateMetrics() {
  metrics.update([
    { label: "Records", value: records.length, note: "Supplied fixture" },
    { label: "Showing", value: filtered().length, note: `of ${records.length} records` },
    { label: "Decisions made", value: records.filter((r) => r.decision !== "Pending").length, note: "Accept or Reject" },
  ])
}
const changed = (rows) => { for (const row of rows) { const original = records.find((r) => r.id === row.id); if (original) original.decision = row.decision } void refreshDetails(); if (chart) void chart.update(chartOptions(filtered())); updateMetrics() }
async function refresh() {
  const rows = filtered()
  if (chart) await chart.update(chartOptions(rows))
  if (table) await table.update({ data: rows, columns, pageSize: 6, rowId: "id", onChange: changed })
  updateMetrics()
}

Atelier.field(review.controls, { label: "Local draft note", type: "textarea", value: draftNote, description: "Stored in this page session and shown in selected record details.", onChange: async (v) => { draftNote = v; await refreshDetails() } })
