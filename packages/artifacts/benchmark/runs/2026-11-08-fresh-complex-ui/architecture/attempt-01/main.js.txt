const iconIds = {
  n1:"aws:amazon-api-gateway", n2:"aws:aws-lambda", n3:"aws:amazon-simple-storage-service",

fixture.nodes = fixture.nodes.map(node => ({...node, icon:iconIds[node.id]}))
const book = Atelier.notebook({title:"Architecture investigation notebook",description:"Illustrative research workspace · supplied fixture · no production evidence"})
const intro = book.cell({id:"brief",title:"Investigation brief"})
const brief = Atelier.stack(intro.view)
Atelier.notice(brief,{title:"Illustrative data",text:"Fixture supplied for UI evaluation. These descriptions and statuses are illustrative; no production evidence or live backend is connected."})
Atelier.mount("markdown",brief,{source:"## Trace a request across the review path\n\nFollow an authenticated request from **API Gateway** through validation, durable queueing, orchestration and inference. The evidence archive, decision store and telemetry complete the review trail.\n\nSelect a service in the workflow or inventory to synchronize its inspector. Use the local note for investigation context, then export the selected service and note."})
Atelier.metrics(brief,[{label:"Services",value:"12",note:"Supplied fixture"},{label:"Connections",value:String(fixture.edges.length),note:"Illustrative paths"},{label:"Warnings",value:String(fixture.nodes.filter(n=>n.status==="Warning").length),note:"Fixture status"}])
const graphCell=book.cell({id:"workflow",title:"AWS workflow map",controls:true})
let selectedId="n1", health="All", flow, summaryHandle, propertiesHandle, exampleHandle, detailViews
let noteValue=""
const current=()=>fixture.nodes.find(n=>n.id===selectedId)
const toneFor=n=>health!=="All"&&n.status!==health?"neutral":n.status==="Warning"?"warning":"success"
const graphData=()=>({nodes:fixture.nodes.map(n=>({id:n.id,selected:n.id===selectedId,data:{label:n.label,description:n.description,status:n.status,tone:toneFor(n),icon:icons.url(n.icon)}})),edges:fixture.edges.map(([a,b])=>({id:`e${a}-${b}`,source:`n${a}`,target:`n${b}`}))})
const propertiesFor=n=>({id:n.id,service:n.service,label:n.label,status:n.status,description:n.description,icon:iconIds[n.id]})
const exampleFor=n=>n.id==="n12"?`SELECT record_id, decision, archived_at\nFROM evidence_audit\nWHERE service_id = '${n.id}'\nORDER BY archived_at DESC;`:JSON.stringify({serviceId:n.id,traceId:"trace-example-7f2a",event:"processing.requested",evidenceVersion:"v1"},null,2)
async function refreshDetails(){const n=current();if(summaryHandle)await summaryHandle.update({source:`## ${n.service}\n\n**${n.status}** · ${n.id}\n\n${n.description}\n\nThis summary reflects the supplied illustrative fixture.`});if(propertiesHandle)await propertiesHandle.update({data:propertiesFor(n)});if(exampleHandle)await exampleHandle.update({language:n.id==="n12"?"sql":"json",source:exampleFor(n)})}
const selectNode=Atelier.field(graphCell.controls,{label:"Selected service",type:"select",value:selectedId,options:fixture.nodes.map(n=>({value:n.id,label:`${n.id} · ${n.service}`})),onChange:id=>void choose(id,true)})
async function choose(id,focus=false){selectedId=id;selectNode.value=id;if(flow){flow.api.flow?.setNodes(nodes=>nodes.map(node=>{const record=fixture.nodes.find(item=>item.id===node.id);return {...node,selected:node.id===id,data:{...node.data,tone:record?toneFor(record):node.data.tone}}}));if(focus)flow.api.focus(id)}await refreshDetails()}
Atelier.field(graphCell.controls,{label:"Health emphasis",type:"select",value:health,options:["All","Ready","Warning"],description:"Highlights matching nodes without hiding the workflow.",onChange:value=>{health=value;flow.api.flow?.setNodes(nodes=>nodes.map(node=>{const record=fixture.nodes.find(item=>item.id===node.id);return {...node,data:{...node.data,tone:record?toneFor(record):node.data.tone}}}))}})
Atelier.button("Fit overview",()=>flow.api.flow?.fitView({padding:.16}),graphCell.controls,{variant:"secondary"})
Atelier.button("Focus selected",()=>flow.api.focus(selectedId),graphCell.controls,{variant:"secondary"})
flow=Atelier.mount("flow",graphCell.view,{...graphData(),focusId:selectedId,onNodeClick:(_event,node)=>void choose(node.id)})
await flow.ready
Atelier.canvasLink(flow,graphData(),{parent:graphCell.header})
const inspector=book.cell({id:"inspector",title:"Service inspector",controls:true})
Atelier.field(inspector.controls,{label:"Local incident note",type:"textarea",value:noteValue,description:"Notebook-local draft. Preserved while switching detail tabs and included in export.",onChange:value=>{noteValue=value}})
Atelier.notice(inspector.controls,{title:"Selection",text:"The inspector follows the selected workflow or inventory row."})
detailViews=Atelier.tabs(inspector.view,{label:"Service details",items:[{id:"summary",label:"Summary",render:panel=>(summaryHandle=Atelier.mount("markdown",panel,{source:""}))},{id:"properties",label:"Properties",render:panel=>(propertiesHandle=Atelier.mount("json",panel,{data:propertiesFor(current())}))},{id:"example",label:"Example",render:panel=>{const n=current();return exampleHandle=Atelier.mount("code",panel,{language:n.id==="n12"?"sql":"json",source:exampleFor(n)})}}]})
Atelier.button("Export selection + note",()=>Atelier.download("architecture-investigation.json",JSON.stringify({selectedId,incidentNote:noteValue},null,2)),inspector.header,{variant:"primary"})
await detailViews.ready;await refreshDetails()
const sequenceCell=book.cell({id:"sequence",title:"Request sequence",controls:true})
const sequenceConfig={source:`sequenceDiagram\n  title: Illustrative request and review sequence\n  participant Client as API client\n  participant Gateway as API Gateway\n  participant Validate as Lambda validator\n  participant Queue as SQS queue\n  participant Workflow as Step Functions\n  participant Model as SageMaker endpoint\n  participant Archive as S3 evidence archive\n  participant State as DynamoDB review state\n  Client->>Gateway: Authenticated request\n  Gateway->>Validate: Route supported API version\n  Validate->>Validate: Validate payload and attach trace id\n  Validate->>Queue: Enqueue accepted record\n  Queue->>Workflow: Start processing\n  Workflow->>Model: Apply supplied classifier\n  Model-->>Workflow: Prediction and model version\n  Workflow->>Archive: Retain input and evidence version\n  Workflow->>State: Persist review decision\n  State-->>Client: Completion available`}
const sequence=Atelier.mount("mermaid",sequenceCell.view,{...sequenceConfig,controls:sequenceCell.controls});await sequence.ready;Atelier.canvasLink(sequence,sequenceConfig,{parent:sequenceCell.header})
const inventory=book.cell({id:"inventory",title:"Service inventory"})
Atelier.mount("table",inventory.view,{data:fixture.nodes.map(({id,service,status})=>({id,service,status})),columns:[{key:"id",title:"ID"},{key:"service",title:"Service"},{key:"status",title:"Health",type:"badge",tone:value=>value==="Ready"?"success":"warning"},{key:"inspect",title:"Inspect",type:"button",action:async row=>{await choose(row.id,true);await detailViews.select("summary");book.go("inspector")}}],pageSize:12})
