import assert from 'node:assert/strict';import {chromium} from '~/Repos/atelier/node_modules/playwright/index.mjs';import {readFile,writeFile} from 'node:fs/promises';
const root='/private/tmp/atelier-fresh-ui-20261108';const browser=await chromium.launch({headless:true,args:['--enable-unsafe-swiftshader']});const results=[];
try{for(const task of ['review','architecture','lab']){const fixture=JSON.parse(await readFile(`${root}/${task}/task.json`,'utf8'));const page=await browser.newPage();await page.goto(`file://${root}/${task}/author/output.html`);await page.waitForFunction(()=>document.documentElement.dataset.artifactReady==='true');await page.evaluate(()=>Atelier.settled());if(task==='architecture'){const actual=await page.evaluate(()=>{const f=Atelier.inspect().find(x=>x.name==='flow').handle.api.flow;return {nodes:f.getNodes().map(n=>({id:n.id,label:n.data.label,description:n.data.description,status:n.data.status})),edges:f.getEdges().map(e=>[Number(e.source.slice(1)),Number(e.target.slice(1))])}});assert.deepEqual(actual.nodes,fixture.data.nodes.map(({id,label,description,status})=>({id,label,description,status})));assert.deepEqual(actual.edges,fixture.data.edges);results.push({task,pass:true,nodes:actual.nodes.length,edges:actual.edges.length})}else{const rows=await page.evaluate(task=>Atelier.inspect().find(x=>x.name==='table'&&(task==='review'||x.handle.host.closest('#observations'))).handle.api.rows,task);const expected=task==='review'?fixture.data:fixture.data.points;assert.equal(rows.length,expected.length);for(const row of expected){const actual=rows.find(r=>r.id===row.id);assert.ok(actual,row.id);for(const [key,value]of Object.entries(row))assert.deepEqual(actual[key],value,`${task}/${row.id}/${key}`)}if(task==='lab'){const surface=await page.evaluate(()=>{const s=Atelier.inspect().find(x=>x.name==='plotly').handle.api.host.data[0];return{x:s.x,y:s.y,z:s.z}});assert.deepEqual(surface,fixture.data.surface)}results.push({task,pass:true,rows:rows.length})}await page.close()}}finally{await browser.close();await writeFile(`${root}/fixture-check.json`,JSON.stringify({results},null,2))}console.log(JSON.stringify(results,null,2));
